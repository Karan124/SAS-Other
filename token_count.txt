# -- Thinking tokens ---------------------------------------------------------
# GPT-5.2 comes in two API variants:
#   - GPT-5.2 Instant  : no thinking tokens (fast, cheaper)
#   - GPT-5.2 Thinking : reasoning model, generates thinking tokens
#
# IMPORTANT: Check with your Azure APIM team which variant is provisioned.
# The model string matters: "gpt-5.2" vs "gpt-5.2-thinking"
#
# Thinking tokens are billed as OUTPUT tokens at $14.00/M.
# They are generated before the final response and appear on your bill
# even though they are not visible in the response text.
#
# For CT classification (closed label set, structured JSON, temperature=0)
# deep reasoning adds little value. If using Thinking mode, set
# reasoning_effort="low" in your API call to minimise thinking tokens.
#
# Estimated thinking tokens by reasoning effort:
#   low    :  ~200-500   tokens per call  (recommended for classification)
#   medium :  ~500-2000  tokens per call
#   high   : ~2000-8000  tokens per call  (overkill for classification)
#
# Set THINKING_TOKENS_PER_CALL = 0 if using GPT-5.2 Instant (no thinking).
# Set to your expected value if using GPT-5.2 Thinking.

                                 # 1000 = Thinking mode medium effort

INPUT_PRICE_USD_PER_M  = 1.75    # USD per 1M input tokens

# ── Run projections ──────────────────────────────────────────────────────────

RUN_PROJECTIONS = [1, 5, 10, 20, 50]

USD_AUD_RATE = 1 / AUD_USD_RATE


def usd_to_aud(v): return v * USD_AUD_RATE
def aud_to_usd(v): return v * AUD_USD_RATE
def fmt_n(n):      return f"{int(n):,}"
def fmt_usd(v):    return f"${v:,.4f}"
def fmt_aud(v):    return f"A${v:,.2f}"
def divider(c="─", w=74): print(c * w)


# ── Token counting ────────────────────────────────────────────────────────────

def get_encoder():
    if not _TIKTOKEN_AVAILABLE:
        return None
    try:
        enc = tiktoken.get_encoding("cl100k_base")
        # Verify it actually works
        enc.encode("test")
        return enc
    except Exception:
        return None


def count_tokens(text: str, enc) -> int:
    """
    Count tokens in a string.
    Uses tiktoken (cl100k_base) if available and working.
    Falls back to word-count approximation (words × 1.3) otherwise.
    Note: cl100k_base is the GPT-4 encoding — very close to GPT-5.2.
    """
    if enc is not None:
        try:
            return len(enc.encode(str(text)))
        except Exception:
            pass
    # Fallback: 1 word ≈ 1.3 tokens
    return max(1, int(len(str(text).split()) * 1.3))


def count_prompt_tokens(enc) -> int:
    """
    Return prompt token count.
    Reads PROMPT_FILE if set, otherwise uses PROMPT_TOKENS_ESTIMATE.
    These tokens are added to EVERY API call since there is no prompt caching.
    """
    if PROMPT_FILE and os.path.exists(PROMPT_FILE):
        with open(PROMPT_FILE, "r", encoding="utf-8") as f:
            content = f.read()
        n = count_tokens(content, enc)
        print(f"  ✓  Prompt tokens read from file '{PROMPT_FILE}': {fmt_n(n)}")
        return n
    elif PROMPT_FILE:
        print(f"  ⚠  Prompt file '{PROMPT_FILE}' not found — using estimate of {fmt_n(PROMPT_TOKENS_ESTIMATE)}")
    else:
        print(f"  ℹ  No prompt file set — using estimate of {fmt_n(PROMPT_TOKENS_ESTIMATE)} prompt tokens per call")
    return PROMPT_TOKENS_ESTIMATE


def count_row_tokens(row: pd.Series, columns: list, enc) -> int:
    """
    Count tokens for a single row using the specified columns.
    Each field is labelled: 'Column Name: value' — matching what you'd send to the LLM.
    """
    parts = []
    for col in columns:
        val = row.get(col, None)
        if val is not None and pd.notna(val) and str(val).strip():
            parts.append(f"{col}: {val}")
    return count_tokens("\n".join(parts), enc)


# ── Cost calculation ──────────────────────────────────────────────────────────

def cost_usd(input_tokens: int, output_tokens: int, thinking_tokens: int = 0) -> float:
    # Thinking tokens are billed as output tokens (same rate)
    total_output = output_tokens + thinking_tokens
    return (
        (input_tokens  / 1_000_000) * INPUT_PRICE_USD_PER_M +
        (total_output  / 1_000_000) * OUTPUT_PRICE_USD_PER_M
    )


# ── Per-file analysis ─────────────────────────────────────────────────────────

def analyse(df: pd.DataFrame, label: str, full_cols: list, minimal_cols: list,
            prompt_tokens: int, enc) -> dict:

    divider("═")
    print(f"  {label.upper()}  —  {len(df):,} records")
    divider("═")

    results = {}

    for profile, cols in [("FULL", full_cols), ("MINIMAL", minimal_cols)]:

        # Validate columns exist in the dataframe
        missing = [c for c in cols if c not in df.columns]
        present = [c for c in cols if c in df.columns]
        if missing:
            print(f"\n  ⚠  [{profile}] Columns not found in file (skipped): {missing}")
        if not present:
            print(f"  ✗  [{profile}] No valid columns — skipping profile.")
            continue

        print(f"\n  Counting tokens — {profile} ({len(present)} columns)…", end="", flush=True)

        data_tokens = [count_row_tokens(row, present, enc) for _, row in df.iterrows()]

        total_data   = sum(data_tokens)
        avg_data     = int(total_data / len(df))
        p50_data     = int(pd.Series(data_tokens).quantile(0.50))
        p95_data     = int(pd.Series(data_tokens).quantile(0.95))
        max_data     = max(data_tokens)

        # Input per call = data tokens + prompt tokens (resent every call — no caching)
        avg_input_per_call = avg_data + prompt_tokens
        p95_input_per_call = p95_data + prompt_tokens

        # Full run totals
        total_input_run  = total_data + (prompt_tokens * len(df))
        total_output_run = OUTPUT_TOKENS_PER_CALL * len(df)

        # Costs
        cost_per_call_usd = cost_usd(avg_input_per_call, OUTPUT_TOKENS_PER_CALL, THINKING_TOKENS_PER_CALL)
        total_thinking_run = THINKING_TOKENS_PER_CALL * len(df)
        cost_per_run_usd  = cost_usd(total_input_run, total_output_run, total_thinking_run)
        cost_per_run_aud  = usd_to_aud(cost_per_run_usd)

        results[profile] = {
            "cols":               len(present),
            "col_names":          present,
            "total_data":         total_data,
            "avg_data":           avg_data,
            "p50_data":           p50_data,
            "p95_data":           p95_data,
            "max_data":           max_data,
            "prompt_tokens":      prompt_tokens,
            "avg_input_per_call": avg_input_per_call,
            "p95_input_per_call": p95_input_per_call,
            "output_per_call":    OUTPUT_TOKENS_PER_CALL,
            "total_input_run":    total_input_run,
            "total_output_run":   total_output_run,
            "total_thinking_run": total_thinking_run,
            "thinking_per_call":  THINKING_TOKENS_PER_CALL,
            "cost_per_call_usd":  cost_per_call_usd,
            "cost_per_run_usd":   cost_per_run_usd,
            "cost_per_run_aud":   cost_per_run_aud,
        }
        print(" done.")

    if not results:
        return {}

    # ── Print tables ──
    profiles = [p for p in ["FULL", "MINIMAL"] if p in results]
    headers  = ["Metric"] + [f"{p} ({results[p]['cols']} cols)" for p in profiles]

    print()
    divider()
    print("  TOKEN BREAKDOWN  (prompt resent every call — no caching)")
    divider()

    def row_vals(key): return [fmt_n(results[p][key]) for p in profiles]

    token_rows = [
        ["Records in file"]              + [fmt_n(len(df))] * len(profiles),
        ["── Data tokens ──────────────"] + [""] * len(profiles),
        ["  Total (all records)"]         + row_vals("total_data"),
        ["  Avg per record"]              + row_vals("avg_data"),
        ["  P50 per record"]              + row_vals("p50_data"),
        ["  P95 per record"]              + row_vals("p95_data"),
        ["  Max (single record)"]         + row_vals("max_data"),
        ["── Per API call ─────────────"] + [""] * len(profiles),
        ["  Prompt tokens (resent each)"] + [fmt_n(results[p]["prompt_tokens"]) for p in profiles],
        ["  Thinking tokens (billed as output)"] + [fmt_n(results[p]["thinking_per_call"]) for p in profiles],
        ["  Avg input (data + prompt)"]   + row_vals("avg_input_per_call"),
        ["  P95 input (data + prompt)"]   + row_vals("p95_input_per_call"),
        ["  Output tokens (estimated)"]   + row_vals("output_per_call"),
        ["── Full run totals ──────────"] + [""] * len(profiles),
        ["  Total input tokens"]          + row_vals("total_input_run"),
        ["  Total output tokens"]         + row_vals("total_output_run"),
        ["  Total thinking tokens"]       + row_vals("total_thinking_run"),
    ]
    print(tabulate(token_rows, headers=headers, tablefmt="simple",
                   colalign=("left",) + ("right",) * len(profiles)))

    print()
    divider()
    print("  COST PER RUN")
    divider()

    cost_rows = [
        ["Cost per call — avg (USD)"]  + [fmt_usd(results[p]["cost_per_call_usd"]) for p in profiles],
        ["Cost per run — full file (USD)"] + [fmt_usd(results[p]["cost_per_run_usd"]) for p in profiles],
        ["Cost per run — full file (AUD)"] + [fmt_aud(results[p]["cost_per_run_aud"]) for p in profiles],
    ]
    print(tabulate(cost_rows, headers=headers, tablefmt="simple",
                   colalign=("left",) + ("right",) * len(profiles)))

    print()
    divider()
    print(f"  RUN PROJECTIONS  (Budget: {fmt_aud(BUDGET_AUD)}  |  1 AUD = {AUD_USD_RATE} USD)")
    divider()

    budget_usd = aud_to_usd(BUDGET_AUD)
    proj_rows = []
    for n in RUN_PROJECTIONS:
        row = [f"{n} run{'s' if n > 1 else ''}"]
        for p in profiles:
            cost_n_aud = results[p]["cost_per_run_aud"] * n
            remaining  = BUDGET_AUD - cost_n_aud
            if remaining >= 0:
                row.append(f"{fmt_aud(cost_n_aud)}  (leaves {fmt_aud(remaining)})")
            else:
                row.append(f"{fmt_aud(cost_n_aud)}  ✗ exceeds budget")
        proj_rows.append(row)

    # Max runs row
    max_runs_row = ["Max runs in budget"]
    for p in profiles:
        max_r = int(budget_usd // results[p]["cost_per_run_usd"])
        max_runs_row.append(fmt_n(max_r))
    proj_rows.append(max_runs_row)

    print(tabulate(proj_rows, headers=headers, tablefmt="simple",
                   colalign=("left",) + ("right",) * len(profiles)))

    return results


# ── Combined ──────────────────────────────────────────────────────────────────

def combined(ctrl_res: dict, proc_res: dict):
    if not ctrl_res or not proc_res:
        return

    divider("═")
    print("  COMBINED — Controls + Processes (one full sprint run)")
    divider("═")

    budget_usd = aud_to_usd(BUDGET_AUD)
    profiles   = [p for p in ["FULL", "MINIMAL"] if p in ctrl_res and p in proc_res]
    headers    = ["Metric"] + profiles

    rows = []
    for p in profiles:
        c_aud = ctrl_res[p]["cost_per_run_aud"]
        pr_aud = proc_res[p]["cost_per_run_aud"]
        comb_aud = c_aud + pr_aud
        comb_usd = aud_to_usd(comb_aud)
        max_r    = int(budget_usd // comb_usd)
        rows.append((p, c_aud, pr_aud, comb_aud, max_r))

    print()
    combo_rows = [
        ["Controls run cost (AUD)"]        + [fmt_aud(r[1]) for r in rows],
        ["Processes run cost (AUD)"]        + [fmt_aud(r[2]) for r in rows],
        ["Combined cost per run (AUD)"]     + [fmt_aud(r[3]) for r in rows],
        ["Max combined runs in budget"]     + [fmt_n(r[4])   for r in rows],
    ]
    print(tabulate(combo_rows, headers=headers, tablefmt="simple",
                   colalign=("left",) + ("right",) * len(profiles)))
