def banner(title: str):
    print()
    print("─" * 70)
    print(f"  {title}")
    print("─" * 70)

def pass_fail(condition: bool) -> str:
    return "✓  PASS" if condition else "✗  FAIL"

def build_control_context(ctrl: dict) -> str:
    """Build the structured control text sent to the LLM."""
    parts = []
    field_map = [
        ("ctrl_id",          "CONTROL ID"),
        ("ctrl_name",        "CONTROL NAME"),
        ("ctrl_objective",   "CONTROL OBJECTIVE"),
        ("ctrl_description", "CONTROL DESCRIPTION"),
        ("test_procedure",   "TEST PROCEDURE / EVIDENCED BY"),
        ("control_type",     "CONTROL TYPE"),
        ("automation_level", "AUTOMATION LEVEL"),
        ("nature",           "PAYMENT NATURE"),
        ("control_frequency","FREQUENCY"),
        ("business_area",    "BUSINESS AREA"),
    ]
    for key, label in field_map:
        val = ctrl.get(key, "")
        if val and val.strip():
            parts.append(f"{label}: {val}")
    return "\n".join(parts)

def get_usage(response) -> dict:
    """Extract token usage from response, including thinking tokens if present."""
    usage = {}
    if hasattr(response, "usage") and response.usage:
        u = response.usage
        usage["input_tokens"]  = getattr(u, "prompt_tokens", None)
        usage["output_tokens"] = getattr(u, "completion_tokens", None)
        usage["total_tokens"]  = getattr(u, "total_tokens", None)
        # Thinking / reasoning tokens (GPT-5 reasoning models)
        details = getattr(u, "completion_tokens_details", None)
        if details:
            usage["reasoning_tokens"] = getattr(details, "reasoning_tokens", None)
    return usage

def save_results(results: dict, run_ts: str):
    """Save results to JSON and readable text summary."""
    os.makedirs(RESULTS_DIR, exist_ok=True)

    json_path = os.path.join(RESULTS_DIR, f"smoke_test_{run_ts}.json")
    txt_path  = os.path.join(RESULTS_DIR, f"smoke_test_{run_ts}.txt")

    # JSON
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(results, f, indent=2, default=str)

    # Readable text summary
    lines = []
    lines.append("=" * 70)
    lines.append("  PAYMENTS CONTROLS POC — SMOKE TEST RESULTS")
    lines.append(f"  Run: {run_ts}")
    lines.append(f"  Model: {CHAT_MODEL}  |  Embedding: {EMBED_MODEL}")
    lines.append(f"  Endpoint: {ENDPOINT}")
    lines.append("=" * 70)

    for test_name, result in results.get("tests", {}).items():
        lines.append("")
        lines.append(f"  {test_name}")
        lines.append("  " + "─" * 60)
        status = result.get("status", "UNKNOWN")
        lines.append(f"  Status   : {status}")
        if result.get("latency_ms"):
            lines.append(f"  Latency  : {result['latency_ms']}ms")
        if result.get("usage"):
            u = result["usage"]
            lines.append(f"  Tokens   : {u.get('input_tokens')} in | "
                         f"{u.get('output_tokens')} out | "
                         f"{u.get('total_tokens')} total"
                         + (f" | {u.get('reasoning_tokens')} thinking"
                            if u.get("reasoning_tokens") else ""))
        if result.get("response_preview"):
            lines.append(f"  Response : {result['response_preview']}")
        if result.get("error"):
            lines.append(f"  Error    : {result['error']}")
        if result.get("notes"):
            for note in result["notes"]:
                lines.append(f"  Note     : {note}")

    lines.append("")
    lines.append("=" * 70)
    passed = sum(1 for r in results.get("tests", {}).values() if r.get("status") == "PASS")
    total  = len(results.get("tests", {}))
    lines.append(f"  RESULT: {passed}/{total} tests passed")
    lines.append("=" * 70)

    with open(txt_path, "w", encoding="utf-8") as f:
        f.write("\n".join(lines))

    print(f"\n  Results saved:")
    print(f"  JSON   → {json_path}")
    print(f"  Text   → {txt_path}")

        result["error"]  = str(e)
        print(f"  {pass_fail(False)}")

def test_2_json_output(client) -> dict:
    banner("TEST 2 — Structured JSON Output")
    print("  Asking model to return a structured JSON classification...")
    result = {"test": "Structured JSON Output"}
    try:
        t0 = time.time()
        response = client.chat.completions.create(
            model=CHAT_MODEL,
            temperature=0,
            max_tokens=200,
            response_format={"type": "json_object"},
            messages=[
                {
                    "role": "system",
                    "content": (
                        "You are a payments risk analyst. "
                        "Always respond with valid JSON only. No prose."
                    ),
                },
                {
                    "role": "user",
                    "content": (
                        "Classify the following control type. "
                        "Return JSON with keys: control_type, confidence (0-1), rationale.\n\n"
                        "Control: A manager reviews and approves all payment batches "
                        "above $50,000 before release to the payment network."
                    ),
                },
            ],
        )
        latency = int((time.time() - t0) * 1000)
        raw     = response.choices[0].message.content.strip()

        # Validate it parses as JSON
        parsed  = json.loads(raw)
        is_json = isinstance(parsed, dict) and len(parsed) > 0

        result["status"]           = "PASS" if is_json else "FAIL"
        result["latency_ms"]       = latency
        result["response_preview"] = raw[:300]
        result["parsed_keys"]      = list(parsed.keys()) if is_json else []
        result["usage"]            = get_usage(response)

        print(f"  {pass_fail(is_json)}  ({latency}ms)")
        print(f"  Parsed keys: {list(parsed.keys())}")
        print(f"  Response   : {raw[:200]}")

    except json.JSONDecodeError as e:
        result["status"] = "FAIL"
        result["error"]  = f"JSON parse error: {e}"
        print(f"  {pass_fail(False)} — response was not valid JSON")
    except Exception as e:
        result["status"] = "FAIL"
        result["error"]  = str(e)
        print(f"  {pass_fail(False)}")
        print(f"  Error : {e}")

    return result

def test_3_reasoning_effort(client) -> dict:
    banner("TEST 3 — Reasoning Effort Comparison (low / medium / high)")
    print("  Classifying your control at three reasoning effort levels...")
    print("  This shows the token and latency trade-off for your environment.")
    result = {"test": "Reasoning Effort Comparison", "efforts": {}}

    ctrl    = TEST_CONTROLS[0]
    context = build_control_context(ctrl)

    system_prompt = (
        "You are a payments risk control analyst for an Australian bank. "
        "You classify controls against a payment risk taxonomy. "
        "Always respond with valid JSON only."
    )

    user_prompt = f"""Classify the following payment control against the CT taxonomy below.

CONTROL:
{context}

CT TAXONOMY:
{MINI_CT_DEFINITIONS}

Return JSON with this exact structure:
{{
  "ctrl_id": "<control id>",
  "ct_classifications": [
    {{
      "ct_id": "<e.g. CT03>",
      "ct_name": "<CT name>",
      "rating": "implements | weakly_implements | does_not_implement",
      "confidence": <0.0 to 1.0>,
      "rationale": "<one sentence citing specific text from the control>"
    }}
  ],
  "notes": "<any relevant observations>"
}}

Only include CT outcomes where there is a defensible relationship."""

    for effort in ["low", "medium", "high"]:
        print(f"\n  Testing reasoning_effort='{effort}'...")
        try:
            kwargs = {
                "model":       CHAT_MODEL,
                "temperature": 0,
                "max_tokens":  1000,
                "response_format": {"type": "json_object"},
                "messages": [
                    {"role": "system", "content": system_prompt},
                    {"role": "user",   "content": user_prompt},
                ],
            }
            # Add reasoning_effort — supported in GPT-5 family
            # Falls back gracefully if not supported by API version
            try:
                kwargs["reasoning_effort"] = effort
            except Exception:
                pass

            t0 = time.time()
            response = client.chat.completions.create(**kwargs)
            latency  = int((time.time() - t0) * 1000)

            raw    = response.choices[0].message.content.strip()
            parsed = json.loads(raw)
            usage  = get_usage(response)

            result["efforts"][effort] = {
                "status":           "PASS",
                "latency_ms":       latency,
                "usage":            usage,
                "response_preview": raw[:400],
                "ct_count":         len(parsed.get("ct_classifications", [])),
            }

            print(f"    {pass_fail(True)}  ({latency}ms)")
            print(f"    Tokens  : {usage.get('input_tokens')} in | "
                  f"{usage.get('output_tokens')} out"
                  + (f" | {usage.get('reasoning_tokens')} thinking"
                     if usage.get("reasoning_tokens") else ""))
            print(f"    CTs found: {len(parsed.get('ct_classifications', []))}")

        except Exception as e:
            result["efforts"][effort] = {"status": "FAIL", "error": str(e)}
            print(f"    {pass_fail(False)}")
            print(f"    Error : {e}")

    result["status"] = (
        "PASS" if all(r.get("status") == "PASS"
                      for r in result["efforts"].values())
        else "PARTIAL"
    )
    return result

def test_4_reproducibility(client) -> dict:
    banner("TEST 4 — Temperature=0 Reproducibility (3 identical calls)")
    print("  Sending the same classification prompt 3 times at temperature=0.")
    print("  All three responses should be identical.")
    result = {"test": "Reproducibility", "runs": []}

    ctrl    = TEST_CONTROLS[0]
    context = build_control_context(ctrl)

    messages = [
        {
            "role": "system",
            "content": (
                "You are a payments risk control analyst. "
                "Always respond with valid JSON only."
            ),
        },
        {
            "role": "user",
            "content": (
                f"Classify this payment control as one of: "
                f"Preventative, Detective, or Corrective. "
                f"Return JSON with keys: classification, confidence, rationale.\n\n"
                f"CONTROL:\n{context}"
            ),
        },
    ]

    responses = []
    for i in range(1, 4):
        try:
            t0 = time.time()
            response = client.chat.completions.create(
                model=CHAT_MODEL,
                temperature=0,
                max_tokens=200,
                response_format={"type": "json_object"},
                messages=messages,
            )
            latency = int((time.time() - t0) * 1000)
            raw = response.choices[0].message.content.strip()
            responses.append(raw)
            result["runs"].append({
                "run":     i,
                "status":  "PASS",
                "latency_ms": latency,
                "response": raw,
                "usage":   get_usage(response),
            })
            print(f"  Run {i}: {pass_fail(True)}  ({latency}ms)  {raw[:80]}…")
        except Exception as e:
            result["runs"].append({"run": i, "status": "FAIL", "error": str(e)})
            print(f"  Run {i}: {pass_fail(False)} — {e}")

    # Check all three are identical
    if len(responses) == 3:
        all_same = responses[0] == responses[1] == responses[2]
        result["all_responses_identical"] = all_same
        result["status"] = "PASS" if all_same else "WARN"
        print(f"\n  All 3 responses identical: {pass_fail(all_same)}")
        if not all_same:
            print("  ⚠  Responses differed — review output in results file.")
            result["notes"] = [
                "Responses were not identical across 3 runs at temperature=0.",
                "This may indicate thinking token variability.",
                "Review the three responses in the JSON results file.",
            ]
    else:
        result["status"] = "FAIL"

    return result

def test_6_token_usage_logging(client) -> dict:
    banner("TEST 6 — Token Usage Logging")
    print("  Sending a classification prompt and inspecting the full usage object.")
    print("  Confirms thinking tokens are visible (critical for cost tracking).")
    result = {"test": "Token Usage Logging"}

    ctrl    = TEST_CONTROLS[0]
    context = build_control_context(ctrl)

    try:
        t0 = time.time()
        response = client.chat.completions.create(
            model=CHAT_MODEL,
            temperature=0,
            max_tokens=300,
            response_format={"type": "json_object"},
            messages=[
                {
                    "role": "system",
                    "content": "You are a payments risk analyst. Respond with JSON only.",
                },
                {
                    "role": "user",
                    "content": (
                        "Classify whether the following control is Preventative, "
                        "Detective, or Corrective, and state which payment lifecycle "
                        "stage it most likely belongs to "
                        "(Initiation/Validation/Authorisation/Transmission/"
                        "Posting/Reporting/Returns). "
                        "Return JSON: classification, lifecycle_stage, confidence, rationale.\n\n"
                        f"CONTROL:\n{context}"
                    ),
                },
            ],
        )
        latency = int((time.time() - t0) * 1000)

        # Inspect raw usage object
        usage_obj = response.usage
        usage = get_usage(response)

        # Log all available usage attributes
        all_attrs = {}
        if usage_obj:
            for attr in dir(usage_obj):
                if not attr.startswith("_"):
                    try:
                        val = getattr(usage_obj, attr)
                        if not callable(val):
                            all_attrs[attr] = val
                    except Exception:
                        pass

        has_input   = usage.get("input_tokens") is not None
        has_output  = usage.get("output_tokens") is not None
        has_thinking= usage.get("reasoning_tokens") is not None

        result["status"]            = "PASS" if (has_input and has_output) else "FAIL"
        result["latency_ms"]        = latency
        result["usage"]             = usage
        result["all_usage_attrs"]   = all_attrs
        result["response_preview"]  = response.choices[0].message.content.strip()[:300]
        result["notes"]             = []

        print(f"  {pass_fail(has_input and has_output)}  ({latency}ms)")
        print(f"  Input tokens   : {usage.get('input_tokens')}  {'✓' if has_input else '✗'}")
        print(f"  Output tokens  : {usage.get('output_tokens')}  {'✓' if has_output else '✗'}")
        print(f"  Thinking tokens: {usage.get('reasoning_tokens') or 'Not returned'}  "
              f"{'✓' if has_thinking else '⚠'}")

        if not has_thinking:
            result["notes"].append(
                "Thinking tokens not returned in usage object. "
                "This may mean gpt-5.4 is in Instant mode (no thinking) "
                "or the API version does not expose reasoning_tokens. "
                "If thinking is expected, confirm with the Azure APIM team "
                "which GPT-5.4 variant is deployed."
            )
            print("  ⚠  Thinking tokens not in response — check notes in results file.")

        print(f"\n  Full usage attributes available: {list(all_attrs.keys())}")

    except Exception as e:
        result["status"] = "FAIL"
        result["error"]  = str(e)
        print(f"  {pass_fail(False)}")
        print(f"  Error : {e}")

    return result
