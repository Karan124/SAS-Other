def build_embedding_text(ctrl: dict) -> str:
    """
    Build text for embedding — free text fields only.
    List/code fields excluded as they add noise, not semantic signal.
    """
    parts = []
    for col in EMBEDDING_COLUMNS:
        val = ctrl.get(col, "").strip()
        if val:
            parts.append(val)
    return " ".join(parts)

def build_llm_context(ctrl: dict) -> str:
    """
    Build labelled context for LLM — all fields included.
    List fields included with their label so the model can interpret them.
    """
    parts = []
    for col in LLM_COLUMNS:
        val = ctrl.get(col, "").strip()
        if val:
            parts.append(f"{col}: {val}")
    return "\n".join(parts)

def get_usage(response) -> dict:
    usage = {}
    if hasattr(response, "usage") and response.usage:
        u = response.usage
        usage["input_tokens"]  = getattr(u, "prompt_tokens", None)
        usage["output_tokens"] = getattr(u, "completion_tokens", None)
        usage["total_tokens"]  = getattr(u, "total_tokens", None)
        details = getattr(u, "completion_tokens_details", None)
        if details:
            usage["reasoning_tokens"] = getattr(details, "reasoning_tokens", None)
    return usage

def load_controls() -> list:
    """Load all controls from Excel. Returns list of dicts."""
    import pandas as pd
    df = pd.read_excel(CONTROLS_FILE)
    controls = []
    for _, row in df.iterrows():
        ctrl = {}
        for col in df.columns:
            val = row.get(col, "")
            ctrl[col] = "" if (val is None or (hasattr(val, "__class__") and
                               val.__class__.__name__ == "float" and
                               val != val)) else str(val).strip()
        controls.append(ctrl)
    return controls

def cosine_similarity(vec_a: list, vec_b: list) -> float:
    dot   = sum(a * b for a, b in zip(vec_a, vec_b))
    mag_a = sum(a ** 2 for a in vec_a) ** 0.5
    mag_b = sum(b ** 2 for b in vec_b) ** 0.5
    if mag_a == 0 or mag_b == 0:
        return 0.0
    return dot / (mag_a * mag_b)

def call_chat(client, model: str, messages: list, supports_temperature: bool = False,
              reasoning_effort: str = "low", max_completion_tokens: int = 1000,
              response_format: dict = None) -> tuple:
    """
    Unified chat call.
    Returns (response_text, usage_dict, latency_ms, error_str)
    """
    kwargs = {
        "model":                 model,
        "messages":              messages,
        "max_completion_tokens": max_completion_tokens,
    }
    if response_format:
        kwargs["response_format"] = response_format
    # Temperature only for non-reasoning models
    if supports_temperature:
        kwargs["temperature"] = 0
    else:
        kwargs["reasoning_effort"] = reasoning_effort

    try:
        t0 = time.time()
        response = client.chat.completions.create(**kwargs)
        latency  = int((time.time() - t0) * 1000)
        text     = response.choices[0].message.content.strip()
        usage    = get_usage(response)
        return text, usage, latency, None
    except Exception as e:
        return None, {}, 0, str(e)

def test_a_grounding(client, controls: list, controls_sparse: dict) -> dict:
    banner("TEST A — Grounded Classification (hallucination control)")
    result = {"test": "Grounded Classification", "sub_tests": {}}

    system_prompt = """You are a payments risk control analyst for an Australian bank.
You classify payment controls against the CT taxonomy below.
STRICT RULES:
1. Classify ONLY based on the text provided. Do not use external knowledge.
2. If the control text does not contain enough evidence to classify confidently, return INSUFFICIENT_INFORMATION.
3. Every rationale MUST include a direct quoted phrase from the control text.
4. Do not infer, assume, or expand beyond what is written.
Always respond with valid JSON only."""

    user_template = """Classify this control against the CT taxonomy.

CONTROL:
{context}

CT TAXONOMY:
{ct_defs}

Return JSON:
{{
  "ctrl_id": "<id>",
  "ct_classifications": [
    {{
      "ct_id": "<e.g. CT3>",
      "ct_name": "<name>",
      "confidence": <0.0-1.0>,
      "rationale": "<must include a quoted phrase from the control text>"
    }}
  ],
  "insufficient_information": <true if not enough evidence>,
  "notes": "<any observations>"
}}"""

    # Sub-test A1: Full control — should classify successfully
    print("\n  A1: Full control — model should classify with quoted evidence...")
    ctx = build_llm_context(controls[0])
    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user",   "content": user_template.format(
            context=ctx, ct_defs=CT_DEFINITIONS)},
    ]
    text, usage, latency, error = call_chat(
        client, PRIMARY_MODEL, messages,
        reasoning_effort="low", max_completion_tokens=1000,
        response_format={"type": "json_object"},
    )
    if error:
        result["sub_tests"]["A1_full_control"] = {"status": "FAIL", "error": error}
        print(f"  A1: {pf(False)} — {error}")
    else:
        try:
            parsed = json.loads(text)
            cts    = parsed.get("ct_classifications", [])
            # Check rationale contains quoted text
            has_quotes = any('"' in c.get("rationale","") for c in cts)
            insufficient = parsed.get("insufficient_information", False)
            passed = len(cts) > 0 and not insufficient
            result["sub_tests"]["A1_full_control"] = {
                "status":      "PASS" if passed else "FAIL",
                "latency_ms":  latency,
                "usage":       usage,
                "ct_count":    len(cts),
                "has_quotes":  has_quotes,
                "insufficient":insufficient,
                "response":    text[:500],
            }
            print(f"  A1: {pf(passed)}  ({latency}ms)")
            print(f"      CTs found: {len(cts)}  |  Quoted evidence: {has_quotes}")
            if not has_quotes:
                print("      ⚠  No quoted phrases in rationale — hallucination risk")
        except json.JSONDecodeError:
            result["sub_tests"]["A1_full_control"] = {"status": "FAIL", "error": "Invalid JSON"}

    # Sub-test A2: Sparse control — model should return INSUFFICIENT_INFORMATION
    print("\n  A2: Sparse control — model should flag INSUFFICIENT_INFORMATION...")
    ctx_sparse = build_llm_context(controls_sparse)
    messages_sparse = [
        {"role": "system", "content": system_prompt},
        {"role": "user",   "content": user_template.format(
            context=ctx_sparse, ct_defs=CT_DEFINITIONS)},
    ]
    text_s, usage_s, latency_s, error_s = call_chat(
        client, PRIMARY_MODEL, messages_sparse,
        reasoning_effort="low", max_completion_tokens=500,
        response_format={"type": "json_object"},
    )
    if error_s:
        result["sub_tests"]["A2_sparse_control"] = {"status": "FAIL", "error": error_s}
        print(f"  A2: {pf(False)} — {error_s}")
    else:
        try:
            parsed_s   = json.loads(text_s)
            insuff     = parsed_s.get("insufficient_information", False)
            cts_s      = parsed_s.get("ct_classifications", [])
            high_conf  = [c for c in cts_s if c.get("confidence", 0) > 0.7]
            passed_s   = insuff or (len(high_conf) == 0)  # either flags or gives low confidence
            result["sub_tests"]["A2_sparse_control"] = {
                "status":       "PASS" if passed_s else "WARN",
                "latency_ms":   latency_s,
                "usage":        usage_s,
                "insufficient_flagged": insuff,
                "high_conf_cts": len(high_conf),
                "response":     text_s[:300],
            }
            verdict = "Correctly flagged uncertainty" if passed_s else "Gave high-confidence answer on sparse data"
            print(f"  A2: {pf(passed_s)}  ({latency_s}ms)")
            print(f"      INSUFFICIENT_INFORMATION flagged: {insuff}")
            print(f"      High-confidence CTs returned: {len(high_conf)}")
            result["sub_tests"]["A2_sparse_control"]["verdict"] = verdict
        except json.JSONDecodeError:
            result["sub_tests"]["A2_sparse_control"] = {"status": "FAIL", "error": "Invalid JSON"}

    a1_ok = result["sub_tests"].get("A1_full_control", {}).get("status") == "PASS"
    a2_ok = result["sub_tests"].get("A2_sparse_control", {}).get("status") in ("PASS", "WARN")
    result["status"] = "PASS" if (a1_ok and a2_ok) else ("WARN" if (a1_ok or a2_ok) else "FAIL")
    return result

        # Build full source text for phrase checking — use actual control from Excel
        ctrl = controls[0]
        source_text = " ".join([
            ctrl.get("Control Name", ""),
            ctrl.get("Control Objective", ""),
            ctrl.get("Control Description", ""),
            ctrl.get("CTRL_EVDNCD", ""),
        ]).lower()

        citation_results = []
        for ct in cts:
            rationale = ct.get("rationale", "")
            # Extract quoted phrases
            quoted = re.findall(r'"([^"]+)"', rationale)
            # Check each quoted phrase appears in source
            found_in_source = []
            for phrase in quoted:
                if phrase.lower() in source_text:
                    found_in_source.append(phrase)
            citation_results.append({
                "ct_id":         ct.get("ct_id"),
                "quoted_phrases":quoted,
                "found_in_source": found_in_source,
                "grounded":      len(found_in_source) > 0,
            })

        all_grounded = all(r["grounded"] for r in citation_results) if citation_results else False
        any_grounded = any(r["grounded"] for r in citation_results) if citation_results else False

        result["status"]           = "PASS" if all_grounded else ("WARN" if any_grounded else "FAIL")
        result["latency_ms"]       = latency
        result["usage"]            = usage
        result["ct_count"]         = len(cts)
        result["citation_results"] = citation_results
        result["all_grounded"]     = all_grounded
        result["response_preview"] = text[:400]
        result["verdict"] = (
            "All rationales grounded in source text" if all_grounded else
            "Some rationales not grounded — possible hallucination" if any_grounded else
            "No rationales grounded in source text — hallucination detected"
        )

        print(f"  {pf(all_grounded)}  ({latency}ms)")
        for r in citation_results:
            icon = "✓" if r["grounded"] else "✗"
            print(f"    {icon} {r['ct_id']}: quoted={r['quoted_phrases']} | "
                  f"found in source={r['found_in_source']}")
        print(f"  Verdict: {result['verdict']}")

    except json.JSONDecodeError:
        result["status"] = "FAIL"
        result["error"]  = "Invalid JSON returned"

    return result

def test_e_reproducibility(client, controls: list) -> dict:
    banner(f"TEST E — Semantic Consistency ({REPRODUCIBILITY_RUNS} runs)")
    print(f"  Running same CT mapping prompt {REPRODUCIBILITY_RUNS} times.")
    print("  Measuring label stability and confidence range.")
    print("  Note: Reasoning models do not support temperature. Variability is expected.")
    result = {"test": "Semantic Consistency", "runs": []}

    ctx = build_llm_context(controls[0])
    messages = [
        {
            "role": "system",
            "content": (
                "You are a payments risk control analyst. "
                "Return only JSON. No prose."
            ),
        },
        {
            "role": "user",
            "content": (
                f"Classify this payment control. Return the single MOST LIKELY CT outcome.\n\n"
                f"CONTROL:\n{ctx}\n\nCT TAXONOMY:\n{CT_DEFINITIONS}\n\n"
                "Return JSON: {\"ct_id\": \"<CT number>\", \"ct_name\": \"<name>\", "
                "\"confidence\": <0.0-1.0>, \"rationale\": \"<one sentence>\"}"
            ),
        },
    ]

    ct_labels    = []
    confidences  = []

    for i in range(1, REPRODUCIBILITY_RUNS + 1):
        text, usage, latency, error = call_chat(
            client, PRIMARY_MODEL, messages,
            reasoning_effort="low", max_completion_tokens=300,
            response_format={"type": "json_object"},
        )
        if error:
            result["runs"].append({"run": i, "status": "FAIL", "error": error})
            print(f"  Run {i:2d}: FAIL — {error}")
            continue

        try:
            parsed = json.loads(text)
            ct_id  = parsed.get("ct_id", "UNKNOWN")
            conf   = parsed.get("confidence", 0)
            ct_labels.append(ct_id)
            confidences.append(conf)
            result["runs"].append({
                "run": i, "status": "PASS", "ct_id": ct_id,
                "confidence": conf, "latency_ms": latency, "usage": usage,
            })
            print(f"  Run {i:2d}: CT={ct_id}  confidence={conf:.2f}  ({latency}ms)")
        except json.JSONDecodeError:
            result["runs"].append({"run": i, "status": "FAIL", "error": "JSON parse error"})

    # Analyse consistency
    if ct_labels:
        from collections import Counter
        label_counts = Counter(ct_labels)
        majority_label, majority_count = label_counts.most_common(1)[0]
        consistency_pct = majority_count / len(ct_labels) * 100
        conf_range = max(confidences) - min(confidences) if confidences else 1
        avg_conf   = sum(confidences) / len(confidences) if confidences else 0

        passed = consistency_pct >= 80 and conf_range <= 0.15

        result["analysis"] = {
            "total_runs":       len(ct_labels),
            "majority_label":   majority_label,
            "majority_count":   majority_count,
            "consistency_pct":  round(consistency_pct, 1),
            "confidence_min":   min(confidences),
            "confidence_max":   max(confidences),
            "confidence_range": round(conf_range, 3),
            "confidence_avg":   round(avg_conf, 3),
            "label_distribution": dict(label_counts),
        }
        result["status"] = "PASS" if passed else ("WARN" if consistency_pct >= 60 else "FAIL")
        result["verdict"] = (
            f"{majority_label} in {majority_count}/{len(ct_labels)} runs "
            f"({consistency_pct:.0f}%) | conf range {conf_range:.2f}"
        )

        print(f"\n  Majority label  : {majority_label} ({majority_count}/{len(ct_labels)} = {consistency_pct:.0f}%)")
        print(f"  Confidence range: {min(confidences):.2f} — {max(confidences):.2f} (spread: {conf_range:.2f})")
        print(f"  Distribution    : {dict(label_counts)}")
        print(f"  {pf(passed)}")
        print("  Pass criteria: ≥80% same label AND confidence range ≤0.15")

    return result

def test_h_embedding_similarity(client, controls: list) -> dict:
    banner(f"TEST H — Embedding Pairwise Similarity ({len(controls)} controls)")
    print("  Embedding all controls and computing pairwise cosine similarity.")
    print("  Ranked pairs help validate that semantically similar controls score higher.")
    print(f"  Embedding model: {EMBED_MODEL}")
    result = {"test": "Embedding Pairwise Similarity", "embeddings": [], "pairs": []}

    # Step 1: Embed all controls
    print(f"\n  Step 1: Computing embeddings for {len(controls)} controls...")
    embeddings = []
    for i, ctrl in enumerate(controls):
        ctrl_id   = ctrl.get("Control ID", f"row_{i}")
        ctrl_name = ctrl.get("Control Name", "")
        emb_text  = build_embedding_text(ctrl)
        if not emb_text.strip():
            print(f"    ⚠  {ctrl_id}: no text to embed — skipping")
            continue
        try:
            response  = client.embeddings.create(model=EMBED_MODEL, input=emb_text)
            vec       = response.data[0].embedding
            token_ct  = getattr(response.usage, "prompt_tokens", 0)
            embeddings.append({
                "ctrl_id":   ctrl_id,
                "ctrl_name": ctrl_name[:60],
                "vector":    vec,
                "tokens":    token_ct,
            })
            print(f"    ✓  [{i+1:2d}/{len(controls)}] {ctrl_id} — {ctrl_name[:50]}")
        except Exception as e:
            print(f"    ✗  {ctrl_id}: {e}")

    result["controls_embedded"] = len(embeddings)
    total_tokens = sum(e["tokens"] for e in embeddings)
    print(f"\n  Embedded: {len(embeddings)}/{len(controls)} controls | Tokens: {total_tokens}")

    if len(embeddings) < 2:
        result["status"] = "FAIL"
        result["error"]  = "Not enough controls embedded to compute similarity"
        return result

    # Step 2: Compute all pairwise cosine similarities
    print("\n  Step 2: Computing pairwise similarity matrix...")
    pairs = []
    n = len(embeddings)
    for i in range(n):
        for j in range(i + 1, n):
            sim = cosine_similarity(embeddings[i]["vector"], embeddings[j]["vector"])
            pairs.append({
                "ctrl_a":    embeddings[i]["ctrl_id"],
                "name_a":    embeddings[i]["ctrl_name"],
                "ctrl_b":    embeddings[j]["ctrl_id"],
                "name_b":    embeddings[j]["ctrl_name"],
                "similarity": round(sim, 4),
            })

    # Step 3: Rank and display
    pairs.sort(key=lambda x: x["similarity"], reverse=True)
    result["pairs"] = pairs

    print(f"\n  TOP 10 MOST SIMILAR pairs (review — do these make sense?):")
    print(f"  {'Sim':<6}  {'Control A':<30}  {'Control B':<30}")
    print(f"  {'─'*6}  {'─'*30}  {'─'*30}")
    for p in pairs[:10]:
        print(f"  {p['similarity']:.4f}  {p['name_a'][:30]:<30}  {p['name_b'][:30]:<30}")

    print(f"\n  BOTTOM 5 LEAST SIMILAR pairs:")
    print(f"  {'Sim':<6}  {'Control A':<30}  {'Control B':<30}")
    print(f"  {'─'*6}  {'─'*30}  {'─'*30}")
    for p in pairs[-5:]:
        print(f"  {p['similarity']:.4f}  {p['name_a'][:30]:<30}  {p['name_b'][:30]:<30}")

    # Stats
    sims = [p["similarity"] for p in pairs]
    avg_sim = sum(sims) / len(sims)
    result["stats"] = {
        "total_pairs":   len(pairs),
        "avg_similarity": round(avg_sim, 4),
        "max_similarity": round(max(sims), 4),
        "min_similarity": round(min(sims), 4),
        "total_tokens":  total_tokens,
    }
    print(f"\n  Stats: {len(pairs)} pairs | avg={avg_sim:.4f} | "
          f"max={max(sims):.4f} | min={min(sims):.4f}")
    print(f"  Total embedding tokens: {total_tokens}")

    result["status"]  = "PASS"
    result["verdict"] = (
        f"All {len(embeddings)} controls embedded. "
        f"Review ranked pairs in results file to validate semantic ordering."
    )
    # Remove vectors from result (too large to save)
    result["embeddings"] = [
        {k: v for k, v in e.items() if k != "vector"} for e in embeddings
    ]
    print(f"\n  {pf(True)} — Review top/bottom pairs above for semantic sanity check.")
    return result
